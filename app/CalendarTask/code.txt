"use client";
import { useState } from "react";

/* ---------------- DATA ---------------- */
const tasks = [
  { id: 1, title: "Daily meeting", date: "2025-12-27", start: "09:00" },
  { id: 2, title: "Design review", date: "2025-12-27", start: "10:30" },
  { id: 3, title: "Code feature", date: "2025-12-27", start: "14:00" },
];

/* ---------------- HELPERS ---------------- */
function formatDate(date) {
  return date.toISOString().split("T")[0];
}

function isBetween(hour, start, end) {
  return hour >= start && hour < end;
}

function getHour(time) {
  return Number(time.split(":")[0]);
}

/* ---------------- COMPONENT ---------------- */
export default function CalendarTaskPage() {
  const [selectedDate] = useState(new Date());
  const selectedString = formatDate(selectedDate);

  const tasksOfDay = tasks.filter(
    (task) => task.date === selectedString
  );

  const hours = Array.from({ length: 24 }, (_, i) => i);

  return (
    <div style={{ padding: "20px", maxWidth: "500px" }}>
      <h2>Agenda</h2>

      {/* TIMELINE */}
      <div>
        {hours.map((hour) => {
          const isFocusTime = isBetween(hour, 9, 11);

          const hourTasks = tasksOfDay.filter(
            (task) => getHour(task.start) === hour
          );

          return (
            <div
              key={hour}
              style={{
                display: "flex",
                minHeight: "60px",
                borderBottom: "1px solid #e5e7eb",
                background: isFocusTime ? "#eff6ff" : "transparent",
              }}
            >
              {/* HOUR LABEL */}
              <div
                style={{
                  width: "60px",
                  fontSize: "12px",
                  color: "#6b7280",
                }}
              >
                {String(hour).padStart(2, "0")}:00
              </div>

              {/* EVENTS */}
              <div style={{ flex: 1, paddingBottom: "10px" }}>
                {hourTasks.map((task) => (
                  <div
                    key={task.id}
                    style={{
                      background: "#2563eb",
                      color: "white",
                      padding: "8px",
                      borderRadius: "8px",
                      marginBottom: "6px",
                      fontSize: "14px",
                    }}
                  >
                    <strong>{task.start}</strong> — {task.title}
                  </div>
                ))}
              </div>
            </div>
          );
        })}
      </div>
    </div>
  );
}
